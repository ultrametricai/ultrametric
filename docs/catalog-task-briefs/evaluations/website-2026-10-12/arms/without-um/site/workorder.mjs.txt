function requiredText(value, label, min, max) {
  if (typeof value !== "string") throw new Error(`${label} is required.`);
  const text = value.trim();
  if (text.length < min) throw new Error(`${label} must contain at least ${min} ${min === 1 ? "character" : "characters"}.`);
  if (text.length > max) throw new Error(`${label} must contain no more than ${max} characters.`);
  return text;
}

// A deterministic website illustration, not an AI service or production integration.
// Preserve the request text; never infer priority, access permission or assignments.
export function buildDraft(input) {
  if (!input || typeof input !== "object") throw new Error("Enter the request details first.");
  const property = requiredText(input.property, "Property", 1, 80);
  const location = requiredText(input.location, "Location", 1, 80);
  const description = requiredText(input.description, "Request", 10, 800);
  if (!Object.hasOwn(categories, input.category)) throw new Error("Choose a request type from the list.");
  return Object.freeze({
    title: categories[input.category],
    property,
    location,
    description,
    status: "Needs your review",
  });
}

function isExampleDomain(hostname) {
  return /(^|\.)(example|invalid|localhost|test)$/.test(hostname) || /(^|\.)example\.(com|net|org)$/.test(hostname);
}

// Configuration should only expose a usable destination, never the fixture inbox.
export function contactDestination(config) {
  if (typeof config.bookingUrl === "string" && config.bookingUrl.trim()) {
    try {
      const url = new URL(config.bookingUrl.trim());
      if (url.protocol === "https:" && url.hostname.includes(".") && !url.username && !url.password && !isExampleDomain(url.hostname)) {
        return { href: url.href, label: "Arrange a conversation", kind: "booking" };
      }
    } catch { /* A missing or invalid destination keeps the honest preview state. */ }
  }
  const email = typeof config.contactEmail === "string" ? config.contactEmail.trim() : "";
  if (/^[A-Z0-9.!#$%&'*+/=?^_`{|}~-]+@[A-Z0-9](?:[A-Z0-9.-]*[A-Z0-9])?\.[A-Z]{2,}$/i.test(email) && !isExampleDomain(email.split("@")[1].toLowerCase())) {
    return {
      href: `mailto:${email}?subject=${encodeURIComponent("A conversation about maintenance requests")}&body=${encodeURIComponent(conversationStarter)}`,
      label: "Start a conversation",
      kind: "email",
    };
  }
  return null;
}
