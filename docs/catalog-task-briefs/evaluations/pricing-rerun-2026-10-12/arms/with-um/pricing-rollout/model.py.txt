#!/usr/bin/env python3
"""Local, synthetic pricing simulation. No network or provider mutations.

Run from any directory: python3 pricing-rollout/model.py
Edit the dated assumptions here only when supported by an updated decision/source.
"""
import calendar
import csv
import hashlib
import json
from datetime import date, timedelta
from decimal import Decimal, ROUND_HALF_UP
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
OUT = Path(__file__).resolve().parent
D = Decimal
AS_OF = date(2026, 10, 12)
ASSUMED_NOTICE = AS_OF  # Scenario input only; actual notice date is unknown.
NEW_BUYER_DATE = date(2026, 11, 1)
FIXED_END = date(2027, 2, 1)
NEW_MONTHLY = {"Team": D("129"), "Business": D("299")}
SOURCES = ["company/company.json", "billing/pricing-proposal.txt",
           "billing/commitments.txt", "billing/subscriptions.csv"]
with (ROOT / "billing/subscriptions.csv").open(newline="") as stream:
    SUBS = list(csv.DictReader(stream))
COMPANY = json.loads((ROOT / "company/company.json").read_text())
SNAPSHOT = COMPANY["snapshotDate"]


def money(value):
    return str(value.quantize(D("0.01"), rounding=ROUND_HALF_UP))


def month_shift(value, months):
    total = value.year * 12 + value.month - 1 + months
    year, month0 = divmod(total, 12)
    month = month0 + 1
    return date(year, month, min(value.day, calendar.monthrange(year, month)[1]))


def current_net(sub, basis="before_discount"):
    amount = D(sub["amount_usd"])
    return amount * (1 - D(sub["discount_percent"]) / 100) if basis == "before_discount" else amount


def target_net(sub):
    amount = NEW_MONTHLY[sub["plan"]] * (10 if sub["period"] == "year" else 1)
    return amount * (1 - D(sub["discount_percent"]) / 100)


def normalize(sub, amount):
    return amount / (12 if sub["period"] == "year" else 1)


def candidate_date(sub):
    if sub["status"] != "active":
        return None
    threshold = max(ASSUMED_NOTICE + timedelta(days=30), NEW_BUYER_DATE)
    if sub["commitment_ref"] == "C-006":
        threshold = max(threshold, FIXED_END)
    first = date.fromisoformat(sub["renewal"])
    step = 12 if sub["period"] == "year" else 1
    offset = 0
    while month_shift(first, offset) < threshold:
        offset += step
    return month_shift(first, offset)


def write_csv(name, rows):
    with (OUT / name).open("w", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def cohort(sub):
    if sub["status"] == "past_due":
        return "hold_delinquency_review"
    if sub["status"] == "cancel_at_period_end":
        return "preserve_pending_cancellation"
    if sub["discount_percent"] != "0":
        return "hold_discount_basis_review"
    if sub["commitment_ref"] == "C-006":
        return "retain_fixed_rate_until_expiry"
    if sub["period"] == "year":
        return "retain_annual_until_term_end"
    return "migrate_at_eligible_renewal"


def note(sub):
    return {
        "S-001": "Nov 3 is too early for a notice first delivered Oct 12; model Dec 3.",
        "S-002": "Retain USD2490 annual price through current term; model USD2990 at Mar 1 renewal.",
        "S-003": "Preserve 20% indefinitely. Listed USD99 gross/net meaning unavailable; both bases modeled. Hold pending confirmation.",
        "S-004": "Past due: no migration or recovery assumed. Existing debt and collection outcome unknown.",
        "S-005": "Preserve cancellation; no price migration, reactivation or new renewal invoice assumed. Nov 20 end date inferred from renewal field; verify.",
        "S-006": "Fixed USD99 until Feb 1; first modeled renewal after expiry is Feb 5. No catch-up charge.",
    }[sub["subscription_id"]]


ledger = []
for sub in SUBS:
    effective = candidate_date(sub)
    ledger.append({
        "subscription_id": sub["subscription_id"], "customer": sub["customer"],
        "source_snapshot_date_assumed": SNAPSHOT, "plan": sub["plan"],
        "currency": "USD", "interval": sub["period"], "lifecycle_status": sub["status"],
        "source_amount_usd": sub["amount_usd"], "amount_basis": "UNKNOWN; both interpretations modeled",
        "quantity": "UNKNOWN; model assumes one flat subscription",
        "discount_percent_retained": sub["discount_percent"],
        "current_net_if_before_discount": money(current_net(sub)),
        "current_net_if_after_discount": money(current_net(sub, "after_discount")),
        "catalog_target_list_usd": money(NEW_MONTHLY[sub["plan"]] * (10 if sub["period"] == "year" else 1)),
        "conditional_target_net_usd": money(target_net(sub)) if effective else "",
        "transition_group": cohort(sub), "source_renewal": sub["renewal"],
        "fixed_rate_protected_until": FIXED_END.isoformat() if sub["commitment_ref"] == "C-006" else "",
        "hypothetical_notice_date": ASSUMED_NOTICE.isoformat() if effective else "",
        "earliest_candidate_effective_date": effective.isoformat() if effective else "",
        "latest_notice_date_for_candidate": (effective - timedelta(days=30)).isoformat() if effective else "",
        "actual_notice_date": "", "notice_evidence": "NONE; not sent",
        "acceptance_requirement": "UNKNOWN; verify from actual terms before execution",
        "recipient_email": "UNKNOWN", "sender_and_support_route": "UNKNOWN",
        "old_provider_price_id": "UNKNOWN", "new_provider_price_id": "UNKNOWN",
        "provider_subscription_item_id": "UNKNOWN", "provider_account_and_mode": "NONE CONNECTED",
        "decision_source": "billing/pricing-proposal.txt; billing/commitments.txt; proposed policy in decision-brief.md",
        "decision_owner": "Pricing owner/founder (name unassigned)",
        "billing_owner": "Unassigned", "policy_approval": "NOT APPROVED",
        "execution_eligible": "NO", "notes": note(sub),
    })
write_csv("customer-transition-ledger.csv", ledger)

prices = []
for plan, amount in NEW_MONTHLY.items():
    for interval in ("month", "year"):
        prices.append({
            "local_mapping_key": f"{plan.lower()}_usd_{interval}_2026_11_proposed",
            "plan": plan, "currency": "USD", "interval": interval,
            "proposed_amount_usd": money(amount * (10 if interval == "year" else 1)),
            "first_new_buyer_date_proposed": NEW_BUYER_DATE.isoformat(),
            "billing_unit": "UNKNOWN; assumes one flat subscription for model",
            "features_and_limits": "Not provided; no changes modeled",
            "old_price_id": "UNKNOWN", "new_test_price_id": "NOT CREATED",
            "new_live_price_id": "NOT CREATED", "state": "LOCAL PROPOSAL ONLY",
        })
write_csv("pricing-matrix.csv", prices)

ACTIVE = [s for s in SUBS if s["status"] == "active"]
ELM = next(s for s in SUBS if s["subscription_id"] == "S-001")
FINAL_DATE = date(2027, 3, 1)


def modeled_mrr(at, basis, scenario):
    total = D(0)
    for sub in ACTIVE:
        effective = candidate_date(sub)
        if scenario == "elm_cancels" and sub["subscription_id"] == "S-001" and at >= effective:
            continue
        migrate = scenario != "hold_current" and at >= effective
        if scenario == "discount_review_stays_held" and sub["subscription_id"] == "S-003":
            migrate = False
        amount = target_net(sub) if migrate else current_net(sub, basis)
        total += normalize(sub, amount)
    return total


scenario_names = ("hold_current", "conditional_all_active_migrate", "discount_review_stays_held", "elm_cancels")
event_dates = sorted({AS_OF, NEW_BUYER_DATE} | {candidate_date(s) for s in ACTIVE})
events = []
monthly = []
summary = {}
for basis in ("before_discount", "after_discount"):
    starting = sum((normalize(s, current_net(s, basis)) for s in ACTIVE), D(0))
    full_list = sum((normalize(s, current_net(s, basis)) for s in SUBS), D(0))
    ending = modeled_mrr(FINAL_DATE, basis, "conditional_all_active_migrate")
    summary[basis] = {
        "active_baseline_mrr_usd": money(starting),
        "all_rows_monthly_equivalent_usd_not_active_mrr": money(full_list),
        "conditional_march_mrr_usd": money(ending),
        "conditional_uplift_mrr_usd": money(ending - starting),
        "conditional_uplift_percent": money((ending - starting) / starting * 100),
        "held_discount_march_mrr_usd": money(modeled_mrr(FINAL_DATE, basis, "discount_review_stays_held")),
        "elm_cancels_march_mrr_usd": money(modeled_mrr(FINAL_DATE, basis, "elm_cancels")),
        "elm_cancels_delta_mrr_usd": money(modeled_mrr(FINAL_DATE, basis, "elm_cancels") - starting),
    }
    for scenario in scenario_names:
        for at in event_dates:
            value = modeled_mrr(at, basis, scenario)
            events.append({"amount_basis": basis, "scenario": scenario, "effective_date": at.isoformat(),
                           "active_cohort_mrr_usd": money(value), "delta_from_active_baseline_usd": money(value - starting),
                           "cash_collected_usd": "UNKNOWN; no billing executed"})
        for offset in range(5):
            month_start = month_shift(NEW_BUYER_DATE, offset)
            next_month = month_shift(month_start, 1)
            invoices = D(0)
            invoice_count = 0
            for sub in ACTIVE:
                first = date.fromisoformat(sub["renewal"])
                step = 12 if sub["period"] == "year" else 1
                k = 0
                while month_shift(first, k) < next_month:
                    due = month_shift(first, k)
                    if due >= month_start:
                        effective = candidate_date(sub)
                        dropped = scenario == "elm_cancels" and sub["subscription_id"] == "S-001" and due >= effective
                        if not dropped:
                            changed = scenario != "hold_current" and due >= effective
                            if scenario == "discount_review_stays_held" and sub["subscription_id"] == "S-003":
                                changed = False
                            invoices += target_net(sub) if changed else current_net(sub, basis)
                            invoice_count += 1
                    k += step
            monthly.append({"amount_basis": basis, "scenario": scenario, "month": month_start.strftime("%Y-%m"),
                            "month_end_active_cohort_mrr_usd": money(modeled_mrr(next_month - timedelta(days=1), basis, scenario)),
                            "scheduled_recurring_invoice_subtotals_usd": money(invoices),
                            "modeled_invoice_count": invoice_count,
                            "cash_collected_usd": "UNKNOWN; no billing executed"})
write_csv("revenue-events.csv", events)
write_csv("monthly-scenarios.csv", monthly)

cohort_rows = []
for basis in ("before_discount", "after_discount"):
    for group in sorted({cohort(s) for s in SUBS}):
        members = [s for s in SUBS if cohort(s) == group]
        cohort_rows.append({"amount_basis": basis, "transition_group": group,
                            "subscription_count": len(members),
                            "snapshot_monthly_equivalent_usd": money(sum((normalize(s, current_net(s, basis)) for s in members), D(0))),
                            "active_baseline_contribution_usd": money(sum((normalize(s, current_net(s, basis)) for s in members if s["status"] == "active"), D(0))),
                            "conditional_march_active_contribution_usd": money(sum((normalize(s, target_net(s)) for s in members if s["status"] == "active"), D(0))),
                            "executed_changes": 0})
write_csv("cohort-reconciliation.csv", cohort_rows)

previews = []
for sub in SUBS:
    effective = candidate_date(sub)
    previews.append({"subscription_id": sub["subscription_id"], "customer": sub["customer"],
                     "artifact_type": "LOCAL SIMULATION; not provider invoice preview",
                     "next_source_renewal": sub["renewal"],
                     "unchanged_net_if_before_discount_usd": money(current_net(sub)),
                     "unchanged_net_if_after_discount_usd": money(current_net(sub, "after_discount")),
                     "conditional_first_increased_invoice_date": effective.isoformat() if effective else "",
                     "conditional_first_increased_invoice_subtotal_usd": money(target_net(sub)) if effective else "",
                     "immediate_charge_intent_usd": "0.00; renewal-only policy, unverified",
                     "proration_intent_usd": "0.00; unverified with provider",
                     "tax_credits_fees_and_final_total": "UNKNOWN; excluded from model",
                     "provider_preview_receipt": "NONE", "cash_collected": "UNKNOWN",
                     "note": note(sub)})
write_csv("invoice-simulations.csv", previews)

execution = [{"subscription_id": s["subscription_id"], "transition_group": cohort(s),
              "requested_live_change": "NONE", "authorized_batch_id": "NONE",
              "execution_status": "PREPARED_ONLY; held from execution", "live_mutation_attempted": "NO",
              "notice_delivery_receipt": "", "approval_scope_reference": "", "provider_preview_id": "",
              "provider_request_id": "", "provider_receipt_id": "", "authoritative_readback": "",
              "observed_invoice_id": "", "observed_invoice_total_usd": "", "observed_payment_status": "",
              "cancellation_or_scheduled_change_readback": "", "retry_required": "NO; no request made"} for s in SUBS]
write_csv("execution-ledger.csv", execution)

# Verification covers the high-impact eligibility, discount and reconciliation rules.
ids = [s["subscription_id"] for s in SUBS]
assert len(ids) == len(set(ids)) == 6, "Input must contain six unique subscriptions"
assert set(ids) == {row["subscription_id"] for row in ledger} == {row["subscription_id"] for row in execution}
expected_dates = {"S-001": "2026-12-03", "S-002": "2027-03-01", "S-003": "2026-12-10", "S-004": None, "S-005": None, "S-006": "2027-02-05"}
assert {s["subscription_id"]: candidate_date(s).isoformat() if candidate_date(s) else None for s in SUBS} == expected_dates
assert summary["before_discount"]["active_baseline_mrr_usd"] == "484.70"
assert summary["after_discount"]["active_baseline_mrr_usd"] == "504.50"
assert summary["before_discount"]["conditional_march_mrr_usd"] == "610.37"
assert summary["before_discount"]["elm_cancels_march_mrr_usd"] == "481.37"
assert summary["before_discount"]["held_discount_march_mrr_usd"] == "586.37"
assert all(row["execution_eligible"] == "NO" for row in ledger)
for basis in ("before_discount", "after_discount"):
    totals = [r for r in cohort_rows if r["amount_basis"] == basis]
    assert sum(r["subscription_count"] for r in totals) == 6
    assert sum(D(r["active_baseline_contribution_usd"]) for r in totals) == D(summary[basis]["active_baseline_mrr_usd"])
    assert sum(D(r["snapshot_monthly_equivalent_usd"]) for r in totals) == D(summary[basis]["all_rows_monthly_equivalent_usd_not_active_mrr"])
assert [r["scheduled_recurring_invoice_subtotals_usd"] for r in monthly if r["amount_basis"] == "before_discount" and r["scenario"] == "conditional_all_active_migrate"] == ["277.20", "331.20", "331.20", "361.20", "3351.20"]

report = {"company": COMPANY["name"], "synthetic": True, "as_of": AS_OF.isoformat(),
          "snapshot_date": SNAPSHOT, "snapshot_date_caveat": "Taken from company record; CSV has no separate as-of timestamp",
          "assumed_notice_date_not_actual": ASSUMED_NOTICE.isoformat(), "live_billing_connected": False,
          "subscription_count": len(SUBS), "active_baseline_count": len(ACTIVE),
          "excluded_baseline_statuses": ["past_due", "cancel_at_period_end"],
          "source_sha256": {name: hashlib.sha256((ROOT / name).read_bytes()).hexdigest() for name in SOURCES},
          "scenarios": summary, "verification": "PASS: IDs, coverage, dates, discounts, cohort totals, monthly billings, execution exclusion",
          "authorization": "No pricing approval, notice send or live billing execution", "currency": "USD",
          "mrr_definition": "Net recurring charge / interval months; active status only; excludes pending cancellation and past due; not GAAP revenue or cash",
          "rounding": "Decimal arithmetic; round displayed totals to cents only after aggregation"}
(OUT / "model-summary.json").write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps({"verification": report["verification"], "scenarios": summary}, indent=2))
