import test from 'node:test';
import assert from 'node:assert/strict';
import { copyFile, mkdtemp, readFile, rm, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const cli = fileURLToPath(new URL('../cli.mjs', import.meta.url));
const example = new URL('../examples/inventory.json', import.meta.url);
const testDirectory = fileURLToPath(new URL('.', import.meta.url));

async function workspace(t) {
  const directory = await mkdtemp(join(testDirectory, '.tmp-cli-'));
  t.after(() => rm(directory, { recursive: true, force: true }));
  return directory;
}

function runCli(cwd, args) {
  const result = spawnSync(process.execPath, [cli, ...args], {
    cwd,
    encoding: 'utf8',
    timeout: 5_000,
  });
  assert.ifError(result.error);
  assert.equal(result.signal, null);
  return result;
}

test('CLI writes the example report using relative paths containing spaces', async t => {
  const cwd = await workspace(t);
  await copyFile(example, join(cwd, 'inventory input.json'));
  await writeFile(join(cwd, 'reorder report.csv'), 'old report\n');

  const result = runCli(cwd, ['inventory input.json', 'reorder report.csv']);

  assert.equal(result.status, 0, result.stderr);
  assert.equal(result.stdout, '');
  assert.equal(result.stderr, '');
  assert.equal(await readFile(join(cwd, 'reorder report.csv'), 'utf8'),
    'sku,available,order\n"brake-pad",6,6\n');
});

test('CLI writes a header-only report for empty inventory', async t => {
  const cwd = await workspace(t);
  await writeFile(join(cwd, 'inventory.json'), '[]');

  const result = runCli(cwd, ['inventory.json', 'report.csv']);

  assert.equal(result.status, 0, result.stderr);
  assert.equal(await readFile(join(cwd, 'report.csv'), 'utf8'), 'sku,available,order\n');
});

for (const args of [[], ['inventory.json']]) {
  test(`CLI exits with usage when given ${args.length} arguments`, async t => {
    const cwd = await workspace(t);
    const result = runCli(cwd, args);

    assert.equal(result.status, 2);
    assert.equal(result.stdout, '');
    assert.match(result.stderr, /Usage: node cli\.mjs INPUT\.json OUTPUT\.csv/);
  });
}

const invalidInputs = [
  { name: 'malformed JSON', contents: '{' },
  { name: 'invalid inventory', contents: '{}' },
  {
    name: 'a reservation exceeding stock',
    contents: '[{"sku":"pad","onHand":1,"reserved":2,"minimum":3,"packSize":1}]',
  },
];

for (const { name, contents } of invalidInputs) {
  test(`CLI preserves an existing report on ${name}`, async t => {
    const cwd = await workspace(t);
    await writeFile(join(cwd, 'inventory.json'), contents);
    await writeFile(join(cwd, 'report.csv'), 'existing report\n');

    const result = runCli(cwd, ['inventory.json', 'report.csv']);

    assert.equal(result.status, 1);
    assert.equal(result.stdout, '');
    assert.notEqual(result.stderr.trim(), '');
    assert.equal(await readFile(join(cwd, 'report.csv'), 'utf8'), 'existing report\n');
  });
}

test('CLI preserves an existing report when the input cannot be read', async t => {
  const cwd = await workspace(t);
  await writeFile(join(cwd, 'report.csv'), 'existing report\n');

  const result = runCli(cwd, ['missing.json', 'report.csv']);

  assert.equal(result.status, 1);
  assert.equal(result.stdout, '');
  assert.match(result.stderr, /ENOENT/);
  assert.equal(await readFile(join(cwd, 'report.csv'), 'utf8'), 'existing report\n');
});

test('CLI reports a write failure when the output parent is missing', async t => {
  const cwd = await workspace(t);
  await copyFile(example, join(cwd, 'inventory.json'));

  const result = runCli(cwd, ['inventory.json', join('missing', 'report.csv')]);

  assert.equal(result.status, 1);
  assert.equal(result.stdout, '');
  assert.match(result.stderr, /ENOENT/);
});
