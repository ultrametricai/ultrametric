import { readFile, writeFile } from 'node:fs/promises';
import { reorderRows, toCsv } from './lib/reorder.mjs';
const [input, output] = process.argv.slice(2);
if (!input || !output) { console.error('Usage: node cli.mjs INPUT.json OUTPUT.csv'); process.exitCode = 2; }
else { try { await writeFile(output, toCsv(reorderRows(JSON.parse(await readFile(input, 'utf8'))))); } catch (error) { console.error(error.message); process.exitCode = 1; } }
