export function reorderRows(items) {
  if (!Array.isArray(items)) throw new TypeError('Inventory must be an array');
  const seen = new Set();
  return items.map(item => {
    if (!item || typeof item.sku !== 'string' || !item.sku.trim() || seen.has(item.sku)) throw new Error('SKU must be nonempty and unique');
    seen.add(item.sku);
    for (const field of ['onHand', 'reserved', 'minimum', 'packSize']) if (!Number.isInteger(item[field]) || item[field] < 0) throw new Error('Counts must be nonnegative integers');
    if (item.packSize === 0 || item.reserved > item.onHand) throw new Error('Invalid pack size or reservation');
    const available = item.onHand - item.reserved;
    const order = Math.ceil(Math.max(0, item.minimum - available) / item.packSize) * item.packSize;
    return { sku: item.sku, available, order };
  }).filter(row => row.order > 0);
}
const quote = value => '"' + String(value).replaceAll('"', '""') + '"';
export function toCsv(rows) { return 'sku,available,order\n' + rows.map(row => [quote(row.sku), row.available, row.order].join(',')).join('\n') + (rows.length ? '\n' : ''); }
