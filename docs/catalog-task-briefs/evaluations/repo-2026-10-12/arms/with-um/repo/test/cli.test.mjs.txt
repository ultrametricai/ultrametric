import test from 'node:test';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { existsSync } from 'node:fs';
import { mkdir, mkdtemp, readFile, rm, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { fileURLToPath } from 'node:url';

const cli = fileURLToPath(new URL('../cli.mjs', import.meta.url));
const example = new URL('../examples/inventory.json', import.meta.url);
const tempRoot = new URL('../.tmp/', import.meta.url);

async function workspace(t) {
  await mkdir(tempRoot, { recursive: true });
  const directory = await mkdtemp(join(fileURLToPath(tempRoot), 'cli-'));
  t.after(() => rm(directory, { recursive: true, force: true }));
  return directory;
}

function run(directory, args) {
  const result = spawnSync(process.execPath, [cli, ...args], {
    cwd: directory,
    encoding: 'utf8',
    timeout: 10_000,
  });
  assert.ifError(result.error);
  assert.equal(result.signal, null, result.stderr);
  assert.equal(result.stdout, '');
  return result;
}

test('CLI writes the example report using paths relative to its working directory', async t => {
  const directory = await workspace(t);
  await writeFile(join(directory, 'inventory.json'), await readFile(example));
  await writeFile(join(directory, 'report.csv'), 'previous report\n');

  const result = run(directory, ['inventory.json', 'report.csv']);

  assert.equal(result.status, 0, result.stderr);
  assert.equal(result.stderr, '');
  assert.equal(await readFile(join(directory, 'report.csv'), 'utf8'),
    'sku,available,order\n"brake-pad",6,6\n');
});

test('CLI writes a header-only report for empty inventory', async t => {
  const directory = await workspace(t);
  await writeFile(join(directory, 'inventory.json'), '[]');

  const result = run(directory, ['inventory.json', 'report.csv']);

  assert.equal(result.status, 0, result.stderr);
  assert.equal(result.stderr, '');
  assert.equal(await readFile(join(directory, 'report.csv'), 'utf8'),
    'sku,available,order\n');
});

test('CLI reports missing arguments with exit code 2', async t => {
  const directory = await workspace(t);
  for (const args of [[], ['inventory.json']]) {
    const result = run(directory, args);
    assert.equal(result.status, 2);
    assert.equal(result.stderr, 'Usage: node cli.mjs INPUT.json OUTPUT.csv\n');
  }
});

for (const [label, input] of [
  ['malformed JSON', '{'],
  ['invalid inventory', '{}'],
]) {
  test(`CLI rejects ${label} without changing an existing report`, async t => {
    const directory = await workspace(t);
    await writeFile(join(directory, 'inventory.json'), input);
    await writeFile(join(directory, 'report.csv'), 'previous report\n');

    const result = run(directory, ['inventory.json', 'report.csv']);

    assert.equal(result.status, 1);
    assert.notEqual(result.stderr.trim(), '');
    assert.equal(await readFile(join(directory, 'report.csv'), 'utf8'),
      'previous report\n');
  });
}

test('CLI reports unreadable input without creating a report', async t => {
  const directory = await workspace(t);

  const result = run(directory, ['missing.json', 'report.csv']);

  assert.equal(result.status, 1);
  assert.match(result.stderr, /ENOENT/);
  assert.equal(existsSync(join(directory, 'report.csv')), false);
});

test('CLI reports a missing output directory with exit code 1', async t => {
  const directory = await workspace(t);
  await writeFile(join(directory, 'inventory.json'), '[]');

  const result = run(directory, ['inventory.json', join('missing', 'report.csv')]);

  assert.equal(result.status, 1);
  assert.match(result.stderr, /ENOENT/);
  assert.equal(existsSync(join(directory, 'missing', 'report.csv')), false);
});
