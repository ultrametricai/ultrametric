#!/usr/bin/env python3
"""Read-only search of the internal support archive; may display private notes."""
import argparse
import json
from pathlib import Path
import sqlite3

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument("query")
parser.add_argument("--archive", type=Path,
                    default=Path(__file__).resolve().parent / "evidence-2026-10-12/archive.sqlite3")
args = parser.parse_args()
db = sqlite3.connect(args.archive.resolve().as_uri() + "?mode=ro", uri=True)
db.row_factory = sqlite3.Row
# Match literal substrings rather than treating user input as SQL or wildcard syntax.
rows = db.execute("""
    SELECT DISTINCT t.* FROM tickets t LEFT JOIN comments c ON c.ticket_id=t.id
    WHERE instr(lower(t.id),lower(?))>0 OR instr(lower(t.subject),lower(?))>0
       OR instr(lower(t.organization),lower(?))>0 OR instr(lower(c.body),lower(?))>0
    ORDER BY t.id
""", [args.query] * 4).fetchall()
result = []
for row in rows:
    ticket = dict(row)
    ticket["comments"] = [dict(c) for c in db.execute("""
        SELECT c.*, a.state AS attachment_state, a.sha256 AS attachment_sha256
        FROM comments c LEFT JOIN attachments a ON a.filename=c.attachment
        WHERE c.ticket_id=? ORDER BY c.id
    """, (row["id"],))]
    result.append(ticket)
db.close()
print(json.dumps({"access": "internal_only_includes_private_notes",
                  "comment_order": "source ID; chronology unavailable",
                  "tickets": result}, indent=2))
