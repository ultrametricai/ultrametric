"""Local file validation and synthetic event replay; no network or sending code.

Run from any directory: python3 /path/to/checks/dry_run.py
The only files written are validation.md and simulated-events.csv beside the drafts.
"""

import csv
from collections import Counter
from dataclasses import dataclass, field
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
with (ROOT / "recipient-ledger.csv").open(newline="") as handle:
    ledger = list(csv.DictReader(handle))
with (ROOT.parent / "contacts.csv").open(newline="") as handle:
    original = list(csv.DictReader(handle))

results = []
events = []


def check(name, condition):
    if not condition:
        raise AssertionError(name)
    results.append(name)


check("Eight unique stable recipient IDs", len(ledger) == 8 and len({r['recipient_id'] for r in ledger}) == 8)
check("One candidate per account; no duplicate nonempty normalized email", len({r['account_id'] for r in ledger}) == 8 and len([r for r in ledger if r['address']]) == len({r['address'].strip().casefold() for r in ledger if r['address']}))
check("Every row has source, date, reason and next action", all(all(r[k] for k in ('source', 'evidence_date', 'reason', 'next_action')) for r in ledger))
check("Every actual row is ineligible and unapproved", all(r['send_eligible'] == r['approved_for_send'] == 'false' for r in ledger))
check("Disposition reconciliation: six held, one suppressed, one existing conversation", Counter(r['disposition'] for r in ledger) == Counter(held=6, suppressed=1, existing_conversation=1))
fixtures = {r['address']: r for r in ledger if r['recipient_id'].startswith('FIX-')}
check("All four original addresses and historical statuses preserved", len(original) == len(fixtures) == 4 and all(r['address'] in fixtures and fixtures[r['address']]['prior_status'] == r['status'] and fixtures[r['address']]['last_activity'] == r['last_activity'] for r in original))
check("Birch remains suppressed", fixtures['team@birch.example']['disposition'] == 'suppressed')
check("Canal remains a separate existing conversation", fixtures['office@canal.example']['disposition'] == 'existing_conversation')
check("Dune remains held after uncertain send", fixtures['hello@dune.example']['disposition'] == 'held')
check("AMP has no guessed email", next(r for r in ledger if r['recipient_id'] == 'PUB-003')['address'] == '')


@dataclass
class SimulatedContact:
    state: str = 'draft'
    seen: set = field(default_factory=set)
    history: list = field(default_factory=list)


STOPPED = {'suppressed', 'declined', 'bounced'}


def apply_event(contact, scenario, kind, event_id, **details):
    """Provider IDs are invented and explicitly marked SIM; not operation receipts."""
    key = ('SIMULATED_PROVIDER', event_id)
    before = contact.state
    if key in contact.seen:
        effect = 'duplicate_ignored'
    else:
        contact.seen.add(key)
        effect = 'processed'
        if kind == 'opt_out':
            contact.state = 'suppressed'
        elif contact.state in STOPPED:
            effect = 'stopped_state_preserved'
        elif kind == 'decline':
            contact.state = 'declined'
        elif kind == 'bounce':
            contact.state = 'bounced'
        elif kind == 'timeout':
            contact.state = 'delivery_hold'
        elif kind == 'delivered' and contact.state in {'draft', 'pending_delivery'}:
            contact.state = 'delivered'
        elif kind == 'substantive_reply':
            contact.state = 'reply_paused'
        elif kind == 'auto_reply' and contact.state in {'draft', 'pending_delivery', 'delivered', 'queued_followup'}:
            contact.state = 'auto_reply_review'
        elif kind == 'offer_slot' and contact.state == 'reply_paused':
            contact.state = 'meeting_proposed'
        elif kind == 'accept_slot' and contact.state == 'meeting_proposed' and details.get('slot_confirmed') and details.get('method_confirmed'):
            contact.state = 'meeting_accepted'
        elif kind == 'attended' and contact.state == 'meeting_accepted':
            contact.state = 'meeting_attended'
        elif kind == 'close_no_reply' and contact.state in {'delivered', 'queued_followup'}:
            contact.state = 'closed_no_reply'
        contact.history.append(kind)
    events.append(dict(scenario=scenario, recipient_id='SIM-' + scenario, provider='SIMULATED_PROVIDER', event_id=event_id, simulated='true', event=kind, before=before, after=contact.state, effect=effect))


def followup_preview(contact, complete=True):
    return 'preview_only' if complete and contact.state in {'delivered', 'queued_followup'} else 'hold'


c = SimulatedContact()
apply_event(c, 'delivery', 'delivered', 'e1')
check("Synthetic delivery permits only a no-send follow-up preview", followup_preview(c) == 'preview_only')
check("Missing required personalization blocks the preview", followup_preview(c, complete=False) == 'hold')
apply_event(c, 'delivery', 'substantive_reply', 'e2')
check("Substantive reply pauses follow-ups", c.state == 'reply_paused' and followup_preview(c) == 'hold')
apply_event(c, 'delivery', 'delivered', 'e3')
check("Late delivery does not restart a replied conversation", c.state == 'reply_paused')
history_length = len(c.history)
apply_event(c, 'delivery', 'substantive_reply', 'e2')
check("Duplicate provider/event ID is processed once", len(c.history) == history_length and events[-1]['effect'] == 'duplicate_ignored')
apply_event(c, 'delivery', 'offer_slot', 'e4')
check("An offered time is only a proposal", c.state == 'meeting_proposed')
apply_event(c, 'delivery', 'accept_slot', 'e5', slot_confirmed=True, method_confirmed=False)
check("Incomplete booking evidence cannot establish acceptance", c.state == 'meeting_proposed')
apply_event(c, 'delivery', 'accept_slot', 'e6', slot_confirmed=True, method_confirmed=True)
check("Explicit slot and method acceptance records accepted, not attended", c.state == 'meeting_accepted')
apply_event(c, 'delivery', 'attended', 'e7')
check("Attendance is a separate event", c.state == 'meeting_attended')

for scenario, event, expected in [('bounce', 'bounce', 'bounced'), ('optout', 'opt_out', 'suppressed'), ('decline', 'decline', 'declined')]:
    c = SimulatedContact(state='queued_followup')
    apply_event(c, scenario, event, scenario + '-1')
    apply_event(c, scenario, 'delivered', scenario + '-2')
    apply_event(c, scenario, 'substantive_reply', scenario + '-3')
    check("Queued follow-up stops on " + scenario + " and late events cannot reopen it", c.state == expected and followup_preview(c) == 'hold')

c = SimulatedContact(state='pending_delivery')
apply_event(c, 'timeout', 'timeout', 't1')
apply_event(c, 'timeout', 'delivered', 't2')
check("Uncertain delivery stays held pending reconciliation; no automatic retry", c.state == 'delivery_hold' and followup_preview(c) == 'hold')
c = SimulatedContact(state='delivered')
apply_event(c, 'auto_reply', 'auto_reply', 'a1')
check("Automated reply pauses for review without establishing interest", c.state == 'auto_reply_review' and followup_preview(c) == 'hold')
c = SimulatedContact(state='delivered')
apply_event(c, 'silence', 'close_no_reply', 's1')
apply_event(c, 'silence', 'delivered', 's2')
check("Closed no-reply sequence cannot restart on late delivery", c.state == 'closed_no_reply' and followup_preview(c) == 'hold')

with (ROOT / 'simulated-events.csv').open('w', newline='') as handle:
    writer = csv.DictWriter(handle, fieldnames=['scenario', 'recipient_id', 'provider', 'event_id', 'simulated', 'event', 'before', 'after', 'effect'])
    writer.writeheader()
    writer.writerows(events)

report = '''# No-send validation

Validation date: October 12, 2026. Run with `python3 sales/first-conversations/checks/dry_run.py` from the workspace root.

**RESULT: PASS — {count} checks.** Eight ledger rows reconcile to zero eligible, zero duplicate, one suppressed, one existing conversation, and six held. The original four source rows and historical statuses match the ledger.

The event replay is a local model of the proposed handling rules. Every event in `simulated-events.csv` uses a SIM recipient and a fictitious provider identifier. Simulated delivery, replies and attendance are not real-world outcomes. No sender, CRM, inbox, booking service, scheduler, authentication, or provider integration was tested. The script contains no network or sending function.

All real ledger rows remain unapproved and ineligible. A preview result inside a synthetic scenario cannot override those values or the evaluation's no-send instruction. Scheduling intervals and recipient timezones are documented proposals, not tested scheduling integrations. An unresolved timeout requires human/provider reconciliation; the model deliberately does not clear that hold on a later event alone.

| Check | Result |
| --- | --- |
{rows}

Local review: personalized claims link to primary sources; unknown building counts remain unknown; required sender fields remain visibly unresolved; the Canal draft answers its existing request; Birch and Dune have no outreach/retry drafts. Live execution and real conversations remain unperformed.
'''.format(count=len(results), rows='\n'.join('| ' + name + ' | PASS |' for name in results))
(ROOT / 'validation.md').write_text(report)
print('PASS: {} checks; {} simulated event rows; zero real sends, bookings, or qualified opportunities.'.format(len(results), len(events)))
