#!/usr/bin/env python3
"""Local, synthetic pricing simulation. Standard library only; no provider calls."""
import argparse
import calendar
import csv
import hashlib
import json
from datetime import date, timedelta
from decimal import Decimal, ROUND_HALF_UP
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
D = Decimal
TODAY = date(2026, 10, 12)
LAUNCH = date(2026, 11, 1)
NEW_MONTHLY = {"Team": D("129"), "Business": D("299")}
GROUPS = {
    "S-001": "standard_notice_dependent",
    "S-002": "annual_term_protected",
    "S-003": "recurring_discount_protected",
    "S-004": "past_due_hold",
    "S-005": "cancellation_excluded",
    "S-006": "fixed_rate_protected",
}
REASONS = {
    "S-001": "Next renewal with at least 30 days notice; Nov 3 is too early if notice is Oct 12 or later.",
    "S-002": "Keep USD 2490 through annual term; next renewal Mar 1 2027, subject to notice.",
    "S-003": "Preserve recurring 20% percentage discount without adding an end date; Nov 10 is too early for Oct 12 notice.",
    "S-004": "Past due: retain current terms and collection state; treatment requires separate decision after review.",
    "S-005": "Preserve cancel_at_period_end; do not renew, increase price, or reverse cancellation.",
    "S-006": "Keep signed USD 99 rate until Feb 1 2027; first modeled renewal after protection is Feb 5, subject to notice.",
}


def money(value):
    return str(D(value).quantize(D("0.01"), rounding=ROUND_HALF_UP))


def shift(anchor, months):
    index = anchor.year * 12 + anchor.month - 1 + months
    year, month0 = divmod(index, 12)
    month = month0 + 1
    return date(year, month, min(anchor.day, calendar.monthrange(year, month)[1]))


def load_rows():
    with (ROOT / "billing/subscriptions.csv").open(newline="") as f:
        rows = list(csv.DictReader(f))
    ids = [r["subscription_id"] for r in rows]
    if len(ids) != len(set(ids)) or set(ids) != set(GROUPS):
        raise ValueError("Snapshot IDs changed: revise the customer policy before regenerating.")
    for r in rows:
        r["old_gross"] = D(r["amount_usd"])
        r["discount"] = D(r["discount_percent"]) / 100
        r["old_net"] = r["old_gross"] * (1 - r["discount"])
        r["new_gross"] = NEW_MONTHLY[r["plan"]] * (10 if r["period"] == "year" else 1)
        r["new_net"] = r["new_gross"] * (1 - r["discount"])
        r["divisor"] = D(12 if r["period"] == "year" else 1)
        r["renewal_date"] = date.fromisoformat(r["renewal"])
    return rows


def eligible_date(row, illustrative_notice):
    if row["status"] != "active":
        return None
    threshold = max(LAUNCH, illustrative_notice + timedelta(days=30))
    if row["commitment_ref"] == "C-006":
        threshold = max(threshold, date(2027, 2, 1))
    step = 12 if row["period"] == "year" else 1
    n = 0
    while shift(row["renewal_date"], n) < threshold:
        n += step
    return shift(row["renewal_date"], n)


def write_csv(path, rows):
    with path.open("w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def generate(out, illustrative_notice):
    out.mkdir(parents=True, exist_ok=True)
    rows = load_rows()
    company = json.loads((ROOT / "company/company.json").read_text())
    snapshot = company["snapshotDate"]
    active = [r for r in rows if r["status"] == "active"]
    baseline = sum((r["old_net"] / r["divisor"] for r in active), D(0))
    eventual = sum((r["new_net"] / r["divisor"] for r in active), D(0))
    delay = max(date(2026, 11, 15), illustrative_notice + timedelta(days=34))
    effects = {r["subscription_id"]: eligible_date(r, illustrative_notice) for r in rows}
    delayed_effects = {r["subscription_id"]: eligible_date(r, delay) for r in rows}
    sources = ["company/company.json", "billing/subscriptions.csv", "billing/commitments.txt", "billing/pricing-proposal.txt"]
    hashes = {p: hashlib.sha256((ROOT / p).read_bytes()).hexdigest() for p in sources}
    revision = hashlib.sha256(json.dumps({"inputs": hashes, "notice_example": str(illustrative_notice), "model": hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}, sort_keys=True).encode()).hexdigest()[:12]
    ledger = []
    execution = []
    for r in rows:
        sid = r["subscription_id"]
        effective = effects[sid]
        ledger.append({
            "model_revision": revision, "snapshot_date": snapshot,
            "subscription_id": sid, "customer": r["customer"], "plan": r["plan"],
            "currency": "USD", "interval": r["period"], "source_status": r["status"],
            "transition_group": GROUPS[sid], "quantity": "", "quantity_model_assumption": "1 subscription; amount treated as whole-period charge",
            "old_price_id": "", "new_test_price_id": "", "new_live_price_id": "",
            "old_list_period_usd_assumed": money(r["old_gross"]), "catalog_proposed_period_usd": money(r["new_gross"]),
            "discount_percent_preserved": r["discount_percent"],
            "current_net_period_usd_modeled": money(r["old_net"]),
            "proposed_net_period_usd_if_later_eligible": money(r["new_net"]) if effective else "",
            "current_normalized_listed_amount_usd": money(r["old_net"] / r["divisor"]),
            "active_baseline_mrr_usd": money(r["old_net"] / r["divisor"]) if effective else "0.00",
            "eventual_mrr_uplift_usd_if_retained": money((r["new_net"]-r["old_net"])/r["divisor"]) if effective else "0.00",
            "source_next_renewal": r["renewal"], "commitment_ref": r["commitment_ref"],
            "actual_notice_at": "", "notice_delivery_receipt": "", "actual_effective_at": "",
            "illustrative_notice_date_only": str(illustrative_notice) if effective else "",
            "illustrative_first_new_price_renewal_only": str(effective) if effective else "",
            "latest_notice_date_for_illustrative_renewal_only": str(effective-timedelta(days=30)) if effective else "",
            "acceptance_requirement": "unverified", "approval_status": "not_approved",
            "release_eligible": "false", "execution_state": "not_executed",
            "policy_reason": REASONS[sid], "policy_source": "billing/commitments.txt; billing/pricing-proposal.txt; billing/subscriptions.csv",
            "decision_owner": "Founder/commercial owner; person unassigned", "billing_owner": "unassigned",
        })
        execution.append({
            "subscription_id": sid, "model_revision": revision, "mode": "local_simulation_only",
            "requested_change": "none", "authorization_ref": "", "authorized_batch_id": "",
            "provider_account": "", "provider_preview_id": "", "provider_request_id": "", "provider_receipt_id": "",
            "notice_receipt_id": "", "observed_price_id": "", "observed_quantity": "",
            "observed_discount": "", "observed_lifecycle_state": "", "observed_scheduled_changes": "",
            "observed_effective_at": "", "observed_invoice_id": "", "observed_invoice_subtotal_usd": "",
            "observed_payment_id": "", "observed_collected_usd": "", "result": "not_executed",
            "hold_reason": "No connected billing account; notice date unavailable; exact batch unapproved. " + REASONS[sid],
        })
    write_csv(out / "transition_ledger.csv", ledger)
    write_csv(out / "execution_ledger.csv", execution)

    invoices = []
    for r in rows:
        sid = r["subscription_id"]
        effective = effects[sid]
        for kind, when, net in [
            ("next_listed_renewal", r["renewal_date"], r["old_net"]),
            ("first_illustrative_new_price_renewal", effective, r["new_net"]),
        ]:
            if kind.startswith("first") and effective is None:
                continue
            if r["status"] == "cancel_at_period_end":
                disposition = "suppressed_if_cancellation_occurs_as_scheduled"
                subtotal = ""
            elif r["status"] == "past_due":
                disposition = "unresolved; no_invoice_prediction_or_collection_assumed"
                subtotal = ""
            else:
                if kind == "next_listed_renewal" and effective <= when:
                    net = r["new_net"]
                disposition = "simulated_recurring_subtotal_only"
                subtotal = money(net)
            invoices.append({
                "model_revision": revision, "subscription_id": sid, "customer": r["customer"],
                "example_type": kind, "illustrative_notice_date": str(illustrative_notice),
                "simulated_renewal_date": str(when), "currency": "USD", "interval": r["period"],
                "old_net_period_usd": money(r["old_net"]), "new_net_period_usd": money(r["new_net"]),
                "discount_percent": r["discount_percent"], "modeled_recurring_subtotal_usd": subtotal,
                "proration_policy_target": "no_midterm_adjustment; change_at_eligible_renewal_only",
                "tax_credits_fees_usage": "unknown; excluded", "actual_invoice_total": "", "cash_collected": "",
                "provider_preview_status": "not_run", "disposition": disposition,
            })
    write_csv(out / "invoice_simulations.csv", invoices)

    scenarios = []
    for month_index in range(5):
        start = shift(LAUNCH, month_index)
        end = date(start.year, start.month, calendar.monthrange(start.year, start.month)[1])
        for name, schedule, churn_id in [
            ("no_price_change_reference", None, None),
            ("proposed_all_active_retained", effects, None),
            ("downside_S001_cancels_at_first_increase", effects, "S-001"),
            ("delayed_notice_all_active_retained", delayed_effects, None),
        ]:
            total = D(0)
            migrated = unchanged = churned = 0
            migrated_mrr = unchanged_mrr = D(0)
            for r in active:
                sid = r["subscription_id"]
                changed = schedule is not None and end >= schedule[sid]
                if changed and sid == churn_id:
                    churned += 1
                    continue
                value = (r["new_net"] if changed else r["old_net"]) / r["divisor"]
                total += value
                if changed:
                    migrated += 1
                    migrated_mrr += value
                else:
                    unchanged += 1
                    unchanged_mrr += value
            assert migrated + unchanged + churned == len(active)
            assert total == migrated_mrr + unchanged_mrr
            scenarios.append({
                "model_revision": revision, "month_end": str(end), "scenario": name,
                "notice_date_example_only": "" if schedule is None else str(delay if schedule is delayed_effects else illustrative_notice),
                "active_cohort_mrr_usd": money(total), "delta_vs_no_change_usd": money(total-baseline),
                "modeled_migrated_count": migrated, "modeled_unchanged_count": unchanged, "modeled_churned_count": churned,
                "migrated_mrr_usd": money(migrated_mrr), "unchanged_mrr_usd": money(unchanged_mrr),
                "held_past_due_count": 1, "excluded_pending_cancel_count": 1,
                "actual_cash_collected": "", "basis": "month_end_run_rate; no_recognized_revenue_or_cash_forecast",
            })
    write_csv(out / "revenue_scenarios.csv", scenarios)
    sum_listed = sum((r["old_net"] / r["divisor"] for r in rows), D(0))
    net_amount_sensitivity = baseline + D("19.80")
    manifest = {
        "company": company["name"], "synthetic": True, "status": "prepared_only",
        "analysis_date": str(TODAY), "snapshot_date_from_company_record": snapshot,
        "subscription_export_timestamp_independently_verified": False,
        "model_revision": revision, "source_sha256": hashes,
        "notice_date_actual": None, "notice_date_example_only": str(illustrative_notice),
        "delayed_notice_example_only": str(delay), "new_buyer_launch_proposed": str(LAUNCH),
        "provider": None, "live_billing_connected": False, "provider_test_performed": False,
        "authorized_subscription_batch": [], "provider_changes_attempted": 0,
        "notices_sent": 0, "catalog_prices_created": 0, "website_published": False,
        "active_baseline_mrr_usd": money(baseline), "all_listed_normalized_net_amount_usd": money(sum_listed),
        "past_due_normalized_usd_excluded": "249.00", "pending_cancel_normalized_usd_excluded": "99.00",
        "eventual_active_mrr_usd_all_retained": money(eventual), "eventual_uplift_usd": money(eventual-baseline),
        "eventual_uplift_percent": money((eventual/baseline-1)*100),
        "downside_eventual_mrr_usd_if_S001_lost": money(eventual-D("129")),
        "amount_basis_sensitivity": {"if_C003_amount_is_already_net_baseline_mrr_usd": money(net_amount_sensitivity), "eventual_uplift_usd_same_new_net": money(eventual-net_amount_sensitivity)},
        "assumptions": [
            "amount_usd is full-period list amount before discount; invoice subtotal rounds to cents.",
            "One subscription unit per row; actual quantity, usage and add-ons are not supplied.",
            "Active-only run-rate metric excludes past_due and cancel_at_period_end; annual net amount divided by 12.",
            "All prices in USD; no CAD conversion; taxes, fees, credits and usage excluded, not treated as zero in actual bills.",
            "30 calendar days modeled from an illustrative notice date, with renewal anchor and lifecycle intent preserved.",
            "Exact contractual timing, notice delivery, acceptance conditions and provider timestamp behavior remain unverified.",
            "All active customers retained in proposal scenario; downside loses S001 at its first increase; no new buyers modeled.",
        ],
        "ultrametric_progress_save": "blocked: approval required but approval policy never; no successful save receipt",
    }
    (out / "model_summary.json").write_text(json.dumps(manifest, indent=2) + "\n")

    notice_parts = [f"""# Customer notice drafts — not approved or sent

Revision `{revision}`. Actual notice date, effective timestamps, recipient emails, sender and support route are unknown. Bracketed fields MUST be resolved before any use. Dates in the ledger are simulations and must not be copied here as approvals. These are four account-specific drafts plus two held-account treatments; no recipient is ready for sending.

For each draft: verify recipient-to-subscription mapping, source invoice amount/discount, exact renewal and notice rules; insert the effective date after documented notice/acceptance conditions can be met; then obtain commercial approval of that exact message. Planned method is renewal billing without customer action, subject to confirming any acceptance requirement. Do not send that wording if acceptance is required.
"""]
    for r in active:
        sid = r["subscription_id"]
        special = ""
        if sid == "S-003":
            special = "Your recurring 20% discount continues with no fixed end date. The Team list price changes from USD 99 to USD 129 per month, making your discounted price USD 103.20 per month."
        if sid == "S-002":
            special = "Your current USD 2,490 annual price remains in place through your current term. This change applies only at an eligible annual renewal."
        if sid == "S-006":
            special = "Your signed USD 99 rate is protected until February 1, 2027. The increase will apply only at an eligible renewal after that protection ends."
        period = "year" if r["period"] == "year" else "month"
        notice_parts.append(f"""## {sid} — {r['customer']}

To: [verified billing contact for {sid}]  
From: [approved Rivet Desk sender]  
Subject: Your Rivet Desk {r['plan']} price at your upcoming renewal

Hello {r['customer']},

We’re updating Rivet Desk pricing. Your {r['plan']} subscription will change from USD {money(r['old_net'])} per {period} to USD {money(r['new_net'])} per {period}, before applicable taxes, starting with your renewal on [verified eligible effective date and timezone]. Your current price remains in place until that renewal.

{special}

[After verifying that no acceptance is required: No action is needed to continue; the updated price will apply at that renewal.] To discuss your plan or cancellation options before the renewal, contact [monitored support email or verified account-management link].

Thank you,  
[approved sender name], Rivet Desk

Internal routing: actual send date [unavailable]; delivery evidence [none]; acceptance [unverified]; owner [unassigned]. No claims about added features or customer results are supported by this snapshot.
""")
    notice_parts.append("""## S-004 — Example Cedar: hold

No price-change notice queued. Resolve delinquency, current invoice state and commercial treatment first. A later approved price change needs a fresh eligible renewal calculation and a customer-specific notice. Do not bundle collection activity with this rollout or assume the old debt is repriced.

## S-005 — Example Harbor: exclude

No price-change notice queued. Preserve pending cancellation. Verify the effective cancellation timestamp (the source lists November 20, 2026 as renewal) before any service-ending message. Do not imply a new recurring charge, reverse cancellation or count this account as pricing uplift.

## Delivery log required for any later release

Record subscription ID, verified recipient, approved copy revision, sender, actual send timestamp, channel, provider message ID, contractually sufficient delivery evidence, any acceptance, effective renewal timestamp and reviewer. Failed, late or disputed notices return that subscription to hold and trigger recalculation.
""")
    (out / "customer_notices.md").write_text("\n".join(notice_parts))

    scenario_lines = ["| Month end | No change | Proposed, retained | Downside: lose Elm | Delayed notice |", "|---|---:|---:|---:|---:|"]
    for month_index in range(5):
        block = scenarios[month_index*4:month_index*4+4]
        scenario_lines.append("| " + block[0]["month_end"] + " | " + " | ".join("$"+v["active_cohort_mrr_usd"] for v in block) + " |")
    ledger_lines = ["| Subscription | Treatment | Current → proposed net charge | Illustrative first increase |", "|---|---|---:|---|"]
    for r in rows:
        sid = r["subscription_id"]
        new = "$" + money(r["new_net"]) if effects[sid] else "held / excluded"
        ledger_lines.append(f"| {sid} {r['customer']} | {GROUPS[sid]} | ${money(r['old_net'])} → {new}/{r['period']} | {effects[sid] or 'None'} |")
    summary = f"""# Rivet Desk pricing rollout — prepared only

Prepared {TODAY}; synthetic company snapshot {snapshot}; model revision `{revision}`. No live billing account is connected (user confirmed). No notice date is available. Prices, customer notices, launch and migration policy remain proposed. No live or test catalog was created, no subscriptions changed, no notices sent and no site published.

The prepared package covers all six source subscriptions exactly once. The currently authorized execution batch is empty. Four active subscriptions are modeled for future renewal changes; one delinquent account is held and one pending cancellation is excluded. ## Proposed pricing

|---|---:|---:|---:|

This file’s primary model uses the pre-discount interpretation.

## Customer transition preview

**All dates below are illustrative, assuming valid notice on {illustrative_notice}. Actual effective dates remain blank in the ledger.** Renewal dates use 30 calendar days, existing billing anchors and the supplied commitments. Exact notice/acceptance rules and timestamps remain to be checked against the actual agreements.

{chr(10).join(ledger_lines)}

For the default October 12 notice example, the 30-day threshold is November 11. Elm’s November 3 and Canal’s November 10 renewals retain old prices. The annual term and Maple’s fixed-price protection are preserved. A later notice can postpone the first increase by a whole renewal period, including a full year for the annual plan. ## Scenario sensitivity

Values are month-end active-cohort MRR, not monthly recognized revenue, invoices or cash. No new sales, expansion, usage, taxes, refunds or payment success are forecast. The proposed scenario is not approved. Downside assumes Elm cancels exactly at its first proposed increase; this is one of four active customers, not a forecast churn rate. {chr(10).join(scenario_lines)}

Losing Elm erases the eventual aggregate uplift: eventual MRR becomes ${money(eventual-D('129'))}, ${money(baseline-(eventual-D('129')))} below the starting active baseline. For homogeneous cohorts, list-price break-even retained revenue is 99/129 = 76.74% for Team and 249/299 = 83.28% for Business. These are algebraic sensitivities, not estimated cancellation probabilities; the six-account ledger is too small to infer demand.

## Reproduce

From workspace root, run `python3 pricing-rollout/model.py`. Default output is `pricing-rollout/generated/`. To compare a different hypothetical notice date without replacing the default, run `python3 pricing-rollout/model.py --illustrative-notice-date 2026-11-15 --output pricing-rollout/scenarios/notice-nov-15`.

The argument is always a simulation input; it does not record delivery or authorize execution. Dates before this analysis date are rejected. Changing source prices, commitments, or cohorts also requires reviewing the policy constants and all handoff/copy files before using a new revision. Run `python3 pricing-rollout/verify_model.py` for the material model checks.

"""
    (out / "rollout_preview.md").write_text(summary)
    return manifest


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--illustrative-notice-date", type=date.fromisoformat, default=TODAY)
    parser.add_argument("--output", type=Path, default=Path(__file__).parent / "generated")
    args = parser.parse_args()
    if args.illustrative_notice_date < TODAY:
        parser.error("Illustrative notice must not predate the analysis date; no backdating.")
    result = generate(args.output, args.illustrative_notice_date)
    print(json.dumps({k: result[k] for k in ("model_revision", "status", "active_baseline_mrr_usd", "eventual_active_mrr_usd_all_retained", "eventual_uplift_usd")}, indent=2))
