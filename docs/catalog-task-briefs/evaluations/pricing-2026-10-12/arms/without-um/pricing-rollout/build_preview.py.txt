#!/usr/bin/env python3
"""Local planning worksheet for the supplied synthetic Rivet Desk snapshot.

Run: python3 pricing-rollout/build_preview.py
Optional: --notice-date YYYY-MM-DD (hypothetical notice delivery date).
Writes only CSV/Markdown files beside this script. No network or billing API.
Contract exceptions below are specific to billing/commitments.txt.
"""

import argparse
import calendar
import csv
from datetime import date, timedelta
from decimal import Decimal, ROUND_HALF_UP
from pathlib import Path


ROOT = Path(__file__).resolve().parent.parent
OUT = Path(__file__).resolve().parent
NEW_MONTHLY = {"Team": Decimal("129"), "Business": Decimal("299")}
AS_OF = date(2026, 10, 12)
NEW_BUYER_LAUNCH = date(2026, 11, 1)
NOTICE_DAYS = 30


def money(value):
    return format(value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP), ".2f")


def renewal_at(anchor, period, offset):
    months = offset * (12 if period == "year" else 1)
    year, month_zero = divmod(anchor.year * 12 + anchor.month - 1 + months, 12)
    month = month_zero + 1
    return date(year, month, min(anchor.day, calendar.monthrange(year, month)[1]))


def eligible_renewal(row, notice):
    threshold = max(notice + timedelta(days=NOTICE_DAYS), NEW_BUYER_LAUNCH)
    if row["commitment_ref"] == "C-006":
        threshold = max(threshold, date(2027, 2, 1))
    anchor = date.fromisoformat(row["renewal"])
    offset = 0
    candidate = anchor
    while candidate < threshold:
        offset += 1
        candidate = renewal_at(anchor, row["period"], offset)
    return candidate


def amounts(row, basis):
    discount = Decimal(1) - Decimal(row["discount_percent"]) / 100
    listed = Decimal(row["amount_usd"])
    current = listed * discount if basis == "before_discount" else listed
    proposed = NEW_MONTHLY[row["plan"]] * (10 if row["period"] == "year" else 1) * discount
    divisor = Decimal(12 if row["period"] == "year" else 1)
    return current, proposed, current / divisor, proposed / divisor


def write_csv(name, rows):
    with (OUT / name).open("w", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--notice-date", type=date.fromisoformat, default=AS_OF)
    args = parser.parse_args()
    if args.notice_date < AS_OF:
        parser.error("No notices have been sent; use a planning date on or after 2026-10-12.")
    with (ROOT / "billing/subscriptions.csv").open(newline="") as handle:
        rows = list(csv.DictReader(handle))
    supported_refs = {"standard", "annual", "C-003", "C-006"}
    supported_statuses = {"active", "past_due", "cancel_at_period_end"}
    for row in rows:
        if row["plan"] not in NEW_MONTHLY or row["period"] not in {"month", "year"}:
            raise ValueError("Unknown plan/period; update the planning worksheet first.")
        if row["commitment_ref"] not in supported_refs or row["status"] not in supported_statuses:
            raise ValueError("Unknown commitment/status; review source notes first.")
        if not Decimal(0) <= Decimal(row["discount_percent"]) <= Decimal(100):
            raise ValueError("Invalid percentage discount.")
    active = [row for row in rows if row["status"] == "active"]
    impact = []
    for basis in ("before_discount", "after_discount"):
        for row in rows:
            current, proposed, old_mrr, new_mrr = amounts(row, basis)
            included = row["status"] == "active"
            effective = eligible_renewal(row, args.notice_date) if included else None
            if row["status"] == "past_due":
                action = "HOLD: resolve payment status and recheck eligibility before planning an increase"
            elif row["status"] == "cancel_at_period_end":
                action = "EXCLUDE: retain cancellation; review anew only if customer resumes"
            else:
                action = "PREVIEW ONLY: requires decision, actual notice evidence, and invoice validation"
            impact.append({
                "synthetic": "true",
                "subscription_id": row["subscription_id"],
                "customer": row["customer"],
                "plan": row["plan"],
                "period": row["period"],
                "currency": "USD",
                "status": row["status"],
                "commitment_ref": row["commitment_ref"],
                "csv_amount_basis_scenario": basis,
                "csv_amount_usd": row["amount_usd"],
                "discount_percent": row["discount_percent"],
                "modeled_current_recurring_amount_usd": money(current),
                "proposed_recurring_amount_usd_if_eligible": money(proposed),
                "potential_delta_per_period_usd": money(proposed - current),
                "included_in_active_forecast": "yes" if included else "no",
                "active_forecast_delta_mrr_usd": money(new_mrr - old_mrr) if included else "0.00",
                "source_next_renewal": row["renewal"],
                "hypothetical_notice_date": args.notice_date.isoformat(),
                "modeled_effective_renewal": effective.isoformat() if effective else "",
                "latest_notice_date_for_modeled_renewal": (effective - timedelta(days=NOTICE_DAYS)).isoformat() if effective else "",
                "action": action,
            })
    write_csv("impact-preview.csv", impact)

    notice_rows = []
    for notice in sorted({args.notice_date, date(2026, 11, 1), date(2026, 11, 4), date(2026, 11, 11)}):
        for row in active:
            effective = eligible_renewal(row, notice)
            notice_rows.append({
                "subscription_id": row["subscription_id"],
                "customer": row["customer"],
                "hypothetical_notice_date": notice.isoformat(),
                "thirty_days_after_notice": (notice + timedelta(days=NOTICE_DAYS)).isoformat(),
                "modeled_effective_renewal": effective.isoformat(),
                "latest_notice_date_for_modeled_renewal": (effective - timedelta(days=NOTICE_DAYS)).isoformat(),
                "state": "SCENARIO ONLY; no notice sent or migration scheduled",
            })
    write_csv("notice-scenarios.csv", notice_rows)

    summary = [
        "# Pricing impact preview",
        "",
        "Synthetic Rivet Desk planning data. Source snapshot: October 10, 2026; planning as of October 12, 2026.",
        "",
        f"Base scenario assumes valid notice on **{args.notice_date}**. This is an illustration, not a selected notice date or proof of delivery. No customer notices have been sent. No live billing account is connected.",
        "",
        "The two amount interpretations are mutually exclusive. Neither is confirmed. Proposed prices are USD, exclude unmodeled taxes/credits/fees, and retain recurring percentage discounts. Annual normalization divides contract value by 12; it is not cash revenue or an accounting revenue schedule.",
        "",
        "## Proposed list prices",
        "",
        "| Plan | Current monthly | Proposed monthly | Change | Proposed annual |",
        "| --- | ---: | ---: | ---: | ---: |",
        "| Team | $99 | $129 | +$30 / 30.30% | $1,290 |",
        "| Business | $249 | $299 | +$50 / 20.08% | $2,990 |",
        "",
        "Annual prices are proposed at 10× monthly. The existing Business annual subscription is $2,490, so its proposed renewal increase is $500 / 20.08%. No current Team annual contract is supplied.",
        "",
        "## Four active subscriptions, excluding payment hold and cancellation",
        "",
        "| CSV amount interpretation | Current normalized MRR | Fully transitioned MRR | Monthly uplift | Uplift % | Annualized uplift |",
        "| --- | ---: | ---: | ---: | ---: | ---: |",
    ]
    for basis in ("before_discount", "after_discount"):
        values = [amounts(row, basis) for row in active]
        old = sum((v[2] for v in values), Decimal(0))
        new = sum((v[3] for v in values), Decimal(0))
        summary.append(f"| {basis.replace('_', ' ')} | ${money(old)} | ${money(new)} | +${money(new-old)} | {money((new-old)/old*100)}% | +${money((new-old)*12)} |")
    summary += [
        "",
        "This is a constant-cohort scenario with no churn, failed payments, plan changes, tax changes, or additional discounts. It does not forecast new-buyer volume. 'Annualized uplift' is the eventual run rate, not first-year cash collected.",
        "",
        "## Customer-level preview",
        "",
        "| Subscription | Treatment | Modeled current → proposed recurring bill | Effective renewal in base scenario | Latest notice date for that renewal |",
        "| --- | --- | --- | --- | --- |",
    ]
    treatment = {"S-001": "Standard monthly", "S-002": "Honor full annual term", "S-003": "Keep 20% recurring discount", "S-004": "Hold while past due (recommended policy)", "S-005": "Exclude scheduled cancellation", "S-006": "Honor $99 fixed rate through commitment"}
    for row in rows:
        gross, proposed, _, _ = amounts(row, "before_discount")
        net, _, _, _ = amounts(row, "after_discount")
        old = f"${money(gross)}" if gross == net else f"${money(gross)} or ${money(net)}"
        bill = f"{old} → ${money(proposed)} / {row['period']}"
        if row["status"] == "active":
            effective = eligible_renewal(row, args.notice_date)
            deadline = effective - timedelta(days=NOTICE_DAYS)
            schedule = f"{effective} | {deadline}"
        else:
            schedule = "Unscheduled | —"
            bill += " (potential only; excluded)"
        summary.append(f"| {row['subscription_id']} — {row['customer']} | {treatment.get(row['subscription_id'], row['commitment_ref'])} | {bill} | {schedule} |")
    summary += [
        "",
        "S-003 has a confirmed 20% recurring percentage discount with no fixed end date. If its CSV $99 is a list amount, the modeled current bill is $79.20 and the increase is $24.00. If $99 is already the discounted bill, the increase to $103.20 is $4.20. The latter interpretation conflicts with the stated $99 Team list price and needs invoice/price-record reconciliation. Do not apply the discount twice or use either current amount in a notice until verified.",
        "",
        "S-004 is not counted as a collectible price increase while past due. S-005 remains canceled at its current period end. The proposed amounts for these two rows are reference values, not migration instructions.",
        "",
        "## Timing sensitivity",
        "",
        "| Hypothetical notice date | Elm (S-001) | Canal (S-003) | Maple (S-006) | Willow annual (S-002) |",
        "| --- | --- | --- | --- | --- |",
    ]
    by_id = {row["subscription_id"]: row for row in active}
    for notice in sorted({args.notice_date, date(2026, 11, 1), date(2026, 11, 4), date(2026, 11, 11)}):
        dates = [str(eligible_renewal(by_id[sid], notice)) for sid in ("S-001", "S-003", "S-006", "S-002")]
        summary.append(f"| {notice} | " + " | ".join(dates) + " |")
    summary += [
        "",
        "Rule: first natural renewal on or after valid notice + 30 calendar days, the proposed November 1 new-buyer launch, and any fixed-price commitment end. Annual subscriptions retain their price through the current term. The launch floor is a conservative planning choice. Renewal anchors are assumed unchanged. Exact notice timing, delivery requirements, provider timezone, renewal timestamps, and proration behavior remain unverified; send with a buffer and validate before scheduling.",
        "",
        "With no notices before October 12, November 3 and November 10 are already too early for an increase. To retain December 3 for Elm, valid notice must occur by November 3; to retain December 10 for Canal, by November 10. Maple's fixed rate ends February 1, 2027; its next modeled natural renewal is February 5, with notice by January 6. Willow's annual renewal is March 1, with notice by January 30.",
        "",
        "## Base-scenario MRR progression",
        "",
        "| Date / event | If amounts are before discounts | If amounts are after discounts |",
        "| --- | ---: | ---: |",
    ]
    effective_dates = {row["subscription_id"]: eligible_renewal(row, args.notice_date) for row in active}
    def mrr_at(when, basis):
        return sum((amounts(row, basis)[3 if effective_dates[row['subscription_id']] <= when else 2] for row in active), Decimal(0))
    for when, event in [(date(2026, 11, 1), "new-buyer target launch")]+[(d, "eligible renewal(s)") for d in sorted(set(effective_dates.values()))]:
        summary.append(f"| {when} — {event} | ${money(mrr_at(when, 'before_discount'))} | ${money(mrr_at(when, 'after_discount'))} |")
    summary += [
        "",
        "The dates and progression above depend on the base notice scenario; changing the actual notice date can defer revenue. All calculations use unrounded Decimal values and round only for display.",
        "",
        "Sources: [company record](../company/company.json), [proposal](../billing/pricing-proposal.txt), [commitments](../billing/commitments.txt), [subscription snapshot](../billing/subscriptions.csv).",
        "",
        "Regenerate this report and both CSVs from the workspace root with `python3 pricing-rollout/build_preview.py --notice-date YYYY-MM-DD`. Recheck source records before adapting this snapshot-specific worksheet.",
    ]
    (OUT / "impact-preview.md").write_text("\n".join(summary) + "\n")
    print(f"Wrote impact-preview.md, impact-preview.csv ({len(impact)} rows), and notice-scenarios.csv ({len(notice_rows)} rows).")


if __name__ == "__main__":
    main()
