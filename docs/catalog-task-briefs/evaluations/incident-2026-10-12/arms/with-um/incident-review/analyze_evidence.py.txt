#!/usr/bin/env python3
"""Rebuild a bounded analysis of the supplied fixture; no network or service access.

Run from any directory: python3 /path/to/incident-review/analyze_evidence.py
Metric windows are unknown. This script deliberately does not sum request counts,
infer continuous alert conditions, or turn reports into affected-user totals.
"""

import csv
import hashlib
import json
import re
from datetime import datetime, timezone
from pathlib import Path
from zoneinfo import ZoneInfo


ROOT = Path(__file__).resolve().parent.parent
OUT = ROOT / "incident-review"
DERIVED = OUT / "derived"
SNAPSHOT = OUT / "evidence-snapshot"
SOURCE_PATHS = [
    "company/company.json",
    "incident/operator-notes.txt",
    "incident/monitoring-config.txt",
    "incident/support-chat.txt",
    "incident/events.jsonl",
    "incident/service-metrics.csv",
]


def parse_utc(value):
    return datetime.fromisoformat(value.replace("Z", "+00:00"))


def utc_string(value):
    return value.astimezone(timezone.utc).isoformat().replace("+00:00", "Z")


def write_json(path, value):
    path.write_text(json.dumps(value, indent=2) + "\n", encoding="utf-8")


def write_csv(path, fields, rows):
    with path.open("w", newline="", encoding="utf-8") as output:
        writer = csv.DictWriter(output, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)


def main():
    DERIVED.mkdir(parents=True, exist_ok=True)
    manifest_entries = []
    for relative in SOURCE_PATHS:
        data = (ROOT / relative).read_bytes()
        target = SNAPSHOT / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        if target.exists() and target.read_bytes() != data:
            raise ValueError(f"Source changed: {relative}; preserve a new version separately")
        if not target.exists():
            target.write_bytes(data)
        manifest_entries.append({
            "source": relative,
            "snapshot": str(target.relative_to(OUT)),
            "bytes": len(data),
            "sha256": hashlib.sha256(data).hexdigest(),
        })
    manifest_path = OUT / "evidence-manifest.json"
    if manifest_path.exists():
        old = json.loads(manifest_path.read_text(encoding="utf-8"))
        if old["files"] != manifest_entries:
            raise ValueError("Manifest differs; retain the existing evidence record")
    else:
        write_json(manifest_path, {
            "captured_at_utc": utc_string(datetime.now(timezone.utc)),
            "kind": "Local copy of supplied synthetic fixture only",
            "live_logs_exported": False,
            "source_snapshot_date": "2026-10-10",
            "requested_live_evidence_window_utc": [
                "2026-10-09T12:45:00Z", "2026-10-09T14:00:00Z"
            ],
            "retention_note": "Seven-day policy supplied; October 9 logs may expire around October 16. Exact deletion behavior unverified.",
            "files": manifest_entries,
        })

    timeline = []

    def add_timeline(ts, kind, details, source, original, zone, note=""):
        timeline.append({
            "timestamp_utc": ts, "kind": kind, "details": details,
            "source": source, "original_timestamp": original,
            "original_timezone": zone, "qualification": note,
        })

    events = []
    with (SNAPSHOT / "incident/events.jsonl").open(encoding="utf-8") as source:
        for line_number, line in enumerate(source, start=1):
            event = json.loads(line)
            events.append(event)
            add_timeline(
                event["ts"], event["event"],
                json.dumps({k: v for k, v in event.items() if k not in {"ts", "event"}}),
                f"incident/events.jsonl:{line_number}", event["ts"], "UTC",
                "Recorded event; no independent clock verification",
            )

    metrics = []
    with (SNAPSHOT / "incident/service-metrics.csv").open(newline="", encoding="utf-8") as source:
        for line_number, row in enumerate(csv.DictReader(source), start=2):
            for field in ["requests", "http_5xx", "db_active_connections", "queue_depth"]:
                row[field] = int(row[field])
                if row[field] < 0:
                    raise ValueError(f"Negative count at metric line {line_number}")
            if not 0 <= row["http_5xx"] <= row["requests"]:
                raise ValueError(f"Invalid request denominator at metric line {line_number}")
            ratio = 100 * row["http_5xx"] / row["requests"] if row["requests"] else None
            row["row_http_5xx_percent"] = round(ratio, 4) if ratio is not None else None
            row["source"] = f"incident/service-metrics.csv:{line_number}"
            row["semantics"] = "Window/type unknown; ratio assumes matching numerator and denominator scope"
            metrics.append(row)
            add_timeline(
                row["timestamp_utc"], "metric_row",
                f"requests={row['requests']}; http_5xx={row['http_5xx']}; row_ratio_percent={row['row_http_5xx_percent']}; db_active_connections={row['db_active_connections']}; queue_depth={row['queue_depth']}",
                row["source"], row["timestamp_utc"], "UTC", row["semantics"],
            )

    support_rows = 0
    for line_number, line in enumerate((SNAPSHOT / "incident/support-chat.txt").read_text(encoding="utf-8").splitlines(), start=1):
        match = re.match(r"^(\d{2}:\d{2}) (.+)$", line)
        if not match:
            continue
        original_time, details = match.groups()
        local = datetime.fromisoformat(f"2026-10-09T{original_time}:00").replace(tzinfo=ZoneInfo("America/New_York"))
        add_timeline(
            utc_string(local), "support_report", details,
            f"incident/support-chat.txt:{line_number}", original_time,
            "America/New_York (UTC-04:00 on 2026-10-09)",
            "Date inferred from October 10 operator note describing yesterday; chat has minute precision",
        )
        support_rows += 1

    timestamps = [row["timestamp_utc"] for row in metrics]
    if timestamps != sorted(set(timestamps)):
        raise ValueError("Metric timestamps are not unique and increasing")
    timeline.sort(key=lambda row: (row["timestamp_utc"], row["source"]))
    write_csv(DERIVED / "metrics.csv", list(metrics[0]), metrics)
    write_csv(DERIVED / "timeline.csv", list(timeline[0]), timeline)
    by_event = {event["event"]: event for event in events}
    first_error = parse_utc(by_event["db_timeout"]["ts"])
    alert = parse_utc(by_event["alert"]["ts"])
    peak = max(metrics, key=lambda row: row["row_http_5xx_percent"])
    summary = {
        "synthetic": True,
        "source_counts": {"files": len(manifest_entries), "events": len(events), "metric_rows": len(metrics), "support_reports": support_rows, "timeline_rows": len(timeline)},
        "metric_coverage_utc": [timestamps[0], timestamps[-1]],
        "metric_window_semantics": "Unknown; cadence of row labels is five minutes, not evidence of five-minute windows",
        "maximum_row_http_5xx_percent": peak["row_http_5xx_percent"],
        "maximum_row_source": peak["source"],
        "maximum_observed_active_connections": max(row["db_active_connections"] for row in metrics),
        "connection_limit": None,
        "maximum_observed_queue_depth": max(row["queue_depth"] for row in metrics),
        "first_recorded_db_timeout_utc": utc_string(first_error),
        "recorded_alert_utc": utc_string(alert),
        "alert_minus_first_recorded_timeout_seconds": int((alert-first_error).total_seconds()),
        "latency_qualification": "Event separation only; not measured customer-impact detection delay",
        "total_requests": None,
        "total_failed_attempts": None,
        "distinct_users_affected": None,
        "actual_impact_start_utc": None,
        "full_recovery_utc": None,
        "outage_duration_seconds": None,
        "aggregation_qualification": "Withheld until metric window/type, overlap, counter resets and endpoint/population scope are established",
        "recovery_qualification": "Lower error row at 13:30Z and two callers report pages loading at 13:31Z; queue remains 84 at 13:30Z; full recovery unverified",
    }
    write_json(DERIVED / "analysis.json", summary)
    print(json.dumps(summary, indent=2))


if __name__ == "__main__":
    main()
