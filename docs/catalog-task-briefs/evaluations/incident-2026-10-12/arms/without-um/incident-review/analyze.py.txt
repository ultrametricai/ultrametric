#!/usr/bin/env python3
"""Reproduce the incident calculations; no network or third-party packages.

CSV interval semantics are unknown. Sums describe supplied rows only, not
distinct customer requests, users, or the complete incident.
"""

import csv
import hashlib
import json
from datetime import datetime
from pathlib import Path
from zoneinfo import ZoneInfo


ROOT = Path(__file__).resolve().parent.parent
OUT = Path(__file__).resolve().parent
SOURCE = ROOT / "incident"
EASTERN = ZoneInfo("America/New_York")


def parse_time(value):
    result = datetime.fromisoformat(value.replace("Z", "+00:00"))
    if result.utcoffset() is None:
        raise ValueError(f"Timestamp lacks timezone: {value}")
    return result


def main():
    with (SOURCE / "service-metrics.csv").open(newline="") as handle:
        rows = list(csv.DictReader(handle))
    if not rows:
        raise ValueError("No metric rows")

    derived = []
    previous = None
    for row in rows:
        when = parse_time(row["timestamp_utc"])
        if previous is not None and when <= previous:
            raise ValueError("Metric timestamps must be strictly increasing")
        previous = when
        values = {key: int(row[key]) for key in (
            "requests", "http_5xx", "db_active_connections", "queue_depth"
        )}
        if min(values.values()) < 0 or values["http_5xx"] > values["requests"]:
            raise ValueError(f"Invalid counts: {row}")
        derived.append({
            "timestamp_utc": row["timestamp_utc"],
            "timestamp_america_new_york": when.astimezone(EASTERN).isoformat(),
            **values,
            "http_5xx_percent": (
                round(100 * values["http_5xx"] / values["requests"], 4)
                if values["requests"] else None
            ),
        })

    with (OUT / "derived-metrics.csv").open("w", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(derived[0]))
        writer.writeheader()
        writer.writerows(derived)

    events = [json.loads(line) for line in
              (SOURCE / "events.jsonl").read_text().splitlines() if line.strip()]
    for event in events:
        event["timestamp_america_new_york"] = (
            parse_time(event["ts"]).astimezone(EASTERN).isoformat()
        )

    sources = sorted(SOURCE.iterdir()) + [ROOT / "company" / "company.json"]
    high_rows = [row for row in derived
                 if row["http_5xx_percent"] is not None
                 and row["http_5xx_percent"] > 10]

    def supplied_row_sums(selected):
        requests = sum(row["requests"] for row in selected)
        errors = sum(row["http_5xx"] for row in selected)
        return {"rows": len(selected), "requests": requests, "http_5xx": errors,
                "ratio_percent": round(100 * errors / requests, 4) if requests else None}

    summary = {
        "scope": "Synthetic supplied incident records only; no live service verification",
        "limitations": [
            "Metric interval length and overlap are unknown; sums may double-count.",
            "HTTP 5xx counts do not establish unique failed requests or affected customers.",
            "Observed maximum connections are not a confirmed capacity limit.",
            "Sparse samples cannot establish exact outage duration or alert correctness.",
            "Recovery followed rollback, pool restart and batch pause; their effects are confounded.",
        ],
        "source_sha256": {
            str(path.relative_to(ROOT)): hashlib.sha256(path.read_bytes()).hexdigest()
            for path in sources if path.is_file()
        },
        "supplied_row_sums_not_incident_totals": supplied_row_sums(derived),
        "rows_above_10_percent_sums_not_incident_totals": supplied_row_sums(high_rows),
        "peak_http_5xx_row": max(derived, key=lambda row: row["http_5xx_percent"]
                                  if row["http_5xx_percent"] is not None else -1),
        "peak_queue_row": max(derived, key=lambda row: row["queue_depth"]),
        "max_observed_active_connections_not_capacity": max(
            row["db_active_connections"] for row in derived),
        "events": events,
    }
    (OUT / "evidence-summary.json").write_text(json.dumps(summary, indent=2) + "\n")
    print("Wrote derived-metrics.csv and evidence-summary.json")
    print("Supplied rows only: 5xx/request ratio; interval semantics remain unknown.")
    for row in derived:
        ratio = row["http_5xx_percent"]
        display = f"{ratio:.2f}%" if ratio is not None else "undefined"
        print(f"{row['timestamp_utc']}  {display}")


if __name__ == "__main__":
    main()
